# Tách từ và dấu câu thành token riêng.
def tokenize(text):
    return re.findall(r"\w+|[^\w\s]", text, flags=re.UNICODE)

# Tìm span thực thể trong danh sách token.
def find_span(tokens, entity):
    entity_tokens = tokenize(entity)
    target = [t.casefold() for t in entity_tokens]
    source = [t.casefold() for t in tokens]

    for start in range(len(source) - len(target) + 1):
        if source[start:start + len(target)] == target:
            return [start, start + len(target) - 1]

    raise ValueError(f"Không tìm thấy thực thể: {entity}")

train_data = []

for sentence, entities in samples:
    tokens = tokenize(sentence)
    ner = []

    for entity, label in entities:
        start, end = find_span(tokens, entity)
        ner.append([start, end, label])

    train_data.append({
        "tokenized_text": tokens,
        "ner": ner
    })

assert len(train_data) == 100

# Kiểm tra và xem thử 2 mẫu đầu.
print("Số lượng mẫu:", len(train_data))
print(json.dumps(train_data[:2], ensure_ascii=False, indent=2))

# Lưu thành file JSON để sử dụng trong pipeline huấn luyện.
with open("legal_ner_train.json", "w", encoding="utf-8") as f:
    json.dump(train_data, f, ensure_ascii=False, indent=2)
